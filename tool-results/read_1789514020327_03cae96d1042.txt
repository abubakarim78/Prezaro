   112→  const [phase, setPhase] = useState<'select' | 'loading' | 'live'>('select')

   114→  const [coursesError, setCoursesError] = useState<string | null>(null)
   115→  const [settings, setSettings] = useState<AppSettings | null>(null)
   116→  const [mode, setMode] = useState<SessionMode>('WALKTHROUGH')
   117→  const [selectedId, setSelectedId] = useState<string | null>(null)
   118→  const [faceCount, setFaceCount] = useState<number | null>(null)
   119→  const [loadStage, setLoadStage] = useState('Preparing…')
   120→  const [bootError, setBootError] = useState<{ title: string; message: string } | null>(null)
   121→  const [live, setLive] = useState<{
   122→    session: Pick<SessionSummary, 'id' | 'courseId' | 'courseCode'> & { mode: SessionMode }
   123→    roster: RosterEntry[]
   124→    checkedEntryIds: string[]
   125→    recent: RecentChip[]
   126→  } | null>(null)
   127→
   128→  const rosterToken = useRef(0)
   129→  const bootToken = useRef(0)
   130→  const engineRef = useRef<FaceApi | null>(null)

   132→  // ---- courses + settings ----
   133→  useEffect(() => {
   134→    let alive = true
   135→    api<{ courses: Course[] }>('/api/courses')
   136→      .then((d) => {
   137→        if (alive) setCourses(d.courses)
   138→      })
   139→      .catch((e) => {
   140→        if (alive) setCoursesError(getErrorMessage(e))
   141→      })
   142→    api<{ settings: AppSettings }>('/api/settings')
   143→      .then((d) => {
   144→        if (alive) {
   145→          setSettings(d.settings)
   146→          setMode(d.settings.defaultMode)
   147→        }
   148→      })
   149→      .catch(() => {})
   150→    return () => {
   151→      alive = false
   152→    }
   153→  }, [])
   154→
   155→  // ---- roster count on course select ----
   156→  const selectCourse = useCallback(
   157→    async (courseId: string) => {
   158→      setSelectedId(courseId)
   159→      setFaceCount(null)
   160→      const course = courses?.find((c) => c.id === courseId)
   161→      const token = ++rosterToken.current
   162→      try {
   163→        const d = await api<{ roster: RosterEntry[] }>(`/api/courses/${courseId}/roster`)
   164→        if (rosterToken.current !== token) return
   165→        setFaceCount(d.roster.filter((r) => r.descriptors.length > 0).length)
   166→        cacheRoster({
   167→          savedAt: new Date().toISOString(),
   168→          course: {
   169→            id: courseId,
   170→            code: course?.code ?? '',
   171→            title: course?.title ?? '',
   172→          },
   173→          roster: d.roster,
   174→        })
   175→      } catch (e) {
   176→      if (rosterToken.current !== token) return
   177→        const cached = getCachedRoster(courseId)
   178→        if (cached) setFaceCount(cached.roster.filter((r) => r.descriptors.length > 0).length)
   179→        else if (e instanceof OfflineError) setFaceCount(-1)

   181→    },
   182→    [courses]
   183→  )

   185→  // ---- session ensure (GET open → POST) ----
   186→  const ensureSession = useCallback(
   187→    async (course: Course, sessionMode: SessionMode): Promise<SessionDetail> => {
   188→      const store = useAppStore.getState()
   189→      if (!store.online) {
   190→        const open = store.openSession
   191→        if (open && open.courseId === course.id) {
   192→          // resume the open session offline
   193→          const pending = pendingForSession(open.id)
   194→          return {
   195→            id: open.id,
   196→            courseId: course.id,
   197→            courseCode: course.code,
   198→            courseTitle: course.title,
   199→            mode: open.mode,
   200→            status: 'OPEN',
   201→            startedAt: new Date().toISOString(),
   202→            presentCount: pending.length,
   203→            rosterSize: course.studentCount,
   204→            lecturerName: store.user?.name ?? '',
   205→            records: pending.map<AttendanceRecord>((p) => ({
   206→              studentId: p.studentId,
   207→              name: p.name,
   208→              status: p.status,
   209→              confidence: p.confidence ?? null,
   210→              markedAt: p.markedAt,
   211→            })),
   212→          }
   213→        }
   214→        throw new Error('Starting a new scan needs a connection — reconnect and try again.')
   215→      }
   216→
   217→      const { sessions } = await api<{ sessions: SessionSummary[] }>(
   218→        `/api/sessions?courseId=${encodeURIComponent(course.id)}&status=OPEN`
   219→      )
   220→      let summary = sessions[0] ?? null
   221→      if (!summary) {
   222→        const created = await api<{ session: SessionSummary }>('/api/sessions', {
   223→          method: 'POST',
   224→          body: { courseId: course.id, mode: sessionMode },
   225→        })
   226→        summary = created.session
   227→      }
   228→      // Pull detail for seeding existing check-ins (best effort)
   229→      try {
   230→        const detail = await api<{ session: SessionDetail }>(`/api/sessions/${summary.id}`)
   231→        return detail.session
   232→      } catch {
   233→        return {
   234→          ...summary,
   235→          lecturerName: store.user?.name ?? '',
   236→          records: [],
   237→        }
   238→      }
   239→    },
   240→    []
   241→  )
   242→
   243→  // ---- start scanning ----
   244→  const startScan = useCallback(async () => {
   245→    const course = courses?.find((c) => c.id === selectedId)
   246→    if (!course) return
   247→    const token = ++bootToken.current
   248→    setPhase('loading')
   249→    setBootError(null)
   250→    setLoadStage('Fetching roster…')
   251→
   252→    try {
   253→      // 1. roster
   254→      let roster: RosterEntry[] | null = null
   255→      try {
   256→        const d = await api<{ course: { id: string; code: string; title: string }; roster: RosterEntry[] }>(
   257→          `/api/courses/${course.id}/roster`
   258→        )
   259→        roster = d.roster
   260→        cacheRoster({
   261→          savedAt: new Date().toISOString(),
   262→          course: d.course,
   263→          roster: d.roster,
   264→        })
   265→      } catch (e) {
   266→        const cached = getCachedRoster(course.id)
   267→        if (cached) roster = cached.roster
   268→        else if (!(e instanceof OfflineError)) throw e
   269→      }
   270→      if (bootToken.current !== token) return
   271→      if (!roster) {
   272→        throw new Error(
   273→          'You are offline and this course has no cached roster. Reconnect and try again.'
   274→        )
   275→      }
   276→
   277→      // 2. face engine
   278→      setLoadStage('Loading face engine…')
   279→      const faceapi = await loadFaceEngine((s) => {
   280→        if (bootToken.current === token) setLoadStage(s)
   281→      })
   282→      if (bootToken.current !== token) return
   283→      engineRef.current = faceapi
   284→
   285→      // 3. session (create / resume)
   286→      setLoadStage('Preparing session…')
   287→      const session = await ensureSession(course, mode)
   288→      if (bootToken.current !== token) return
   289→
   290→      // 4. seed existing check-ins (resume case)
   291→      const byStudent = new Map(roster.map((r) => [r.studentId, r]))
   292→      const checkedEntryIds = new Set<string>()
   293→      const recent: RecentChip[] = []
   294→      const records = [...session.records].sort(
   295→        (a, b) => new Date(b.markedAt).getTime() - new Date(a.markedAt).getTime()
   296→      )
   297→      for (const rec of records) {
   298→        if (rec.status === 'ABSENT') continue
   299→        const entry = byStudent.get(rec.studentId)
   300→        if (!entry) continue
   301→        checkedEntryIds.add(entry.id)
   302→        if (recent.length < 8) {
   303→          recent.push({
   304→            key: `${entry.id}-${rec.markedAt}`,
   305→            entryId: entry.id,
   306→            name: rec.name ?? nameOf(entry),
   307→            time: format(new Date(rec.markedAt), 'h:mm a'),
   308→          })
   309→        }
   310→      }
   311→
   312→      useAppStore.getState().setOpenSession({
   313→        id: session.id,
   314→        courseId: course.id,
   315→        courseCode: course.code,
   316→        mode: mode === 'KIOSK' ? 'KIOSK' : 'WALKTHROUGH',
   317→      })
   318→
   319→      setLive({
   320→        session: {
   321→          id: session.id,
   322→          courseId: course.id,
   323→          courseCode: course.code,
   324→          mode: mode === 'KIOSK' ? 'KIOSK' : 'WALKTHROUGH',
   325→        },
   326→        roster,
   327→        checkedEntryIds: [...checkedEntryIds],
   328→        recent,
   329→      })
   330→      setLoadStage('Starting camera…')
   331→      setPhase('live')
   332→    } catch (e) {
   333→      if (bootToken.current !== token) return
   334→      setBootError({
   335→        title: 'Could not start scanning',
   336→        message: getErrorMessage(e),
   337→      })
   338→    }
   339→  }, [courses, selectedId, mode, ensureSession])
   340→
   341→  // ---- render ----
   342→  if (phase === 'live' && live) {
   343→    return (
   344→      <LiveScreen
   345→        session={live.session}
   346→        roster={live.roster}
   347→        initialChecked={live.checkedEntryIds}
   348→        initialRecent={live.recent}
   349→        liveness={settings?.liveness ?? true}
   350→        engine={engineRef.current}
   351→      />
   352→    )
   353→  }
   354→
   355→  if (phase === 'loading') {
   356→    return (
   357→      <div className="flex h-dvh flex-col items-center justify-center gap-4 bg-background px-6 text-center">
   358→        <div className="relative flex h-16 w-16 items-center justify-center">
   359→          <span className="absolute inset-0 animate-ping rounded-full bg-primary/20" />
   360→          <span className="flex h-14 w-14 items-center justify-center rounded-2xl bg-primary text-primary-foreground shadow-lg shadow-primary/25">
   361→            <ScanFace className="h-7 w-7" />
   362→          </span>
   363→        </div>
   364→        <p className="text-sm font-medium text-muted-foreground" aria-live="polite">
   365→          {loadStage}
   366→        </p>
   367→        {bootError && (
   368→          <div className="w-full max-w-sm rounded-2xl border bg-card p-4 text-left">
   369→            <p className="font-semibold">{bootError.title}</p>
   370→            <p className="mt-1 text-sm text-muted-foreground">{bootError.message}</p>
   371→            <div className="mt-4 flex gap-2">
   372→              <Button className="flex-1" onClick={startScan}>
   373→                <RefreshCw className="h-4 w-4" /> Retry
   374→              </Button>
   375→              <Button variant="outline" onClick={() => setPhase('select')}>
   376→                Back
   377→              </Button>
   378→            </div>
   379→          </div>
   380→        )}
   381→      </div>
   382→    )
   383→  }
   384→
   385→  // ---- select ----
   386→  const selected = courses?.find((c) => c.id === selectedId) ?? null
   387→  return (
   388→    <div className="flex h-dvh flex-col bg-background">
   389→      <header className="flex items-center gap-3 px-4 pt-safe lg:px-8" style={{ paddingTop: 'max(env(safe-area-inset-top), 14px)', paddingBottom: 12 }}>
   390→        <Button
   391→          variant="ghost"
   392→          size="icon"
   393→          className="size-9"
   394→          onClick={() => (useAppStore.getState().openSession ? replaceHomeSafely() : useAppStore.getState().back())}
   395→          aria-label="Close scanner"
   396→        >
   397→          <X className="h-5 w-5" />
   398→        </Button>
   399→        <div>
   400→          <h1 className="text-lg font-bold tracking-tight">Take attendance</h1>
   401→          <p className="text-xs text-muted-foreground">Choose a course and scan mode</p>
   402→        </div>
   403→      </header>
   404→
   405→      <main className="flex-1 overflow-y-auto scrollbar-thin px-4 pb-6">
   406→        {coursesError && (
   407→          <div className="rounded-2xl border bg-card p-4 text-sm">
   408→            <p className="font-semibold">Couldn&apos;t load courses</p>
   409→            <p className="mt-1 text-muted-foreground">{coursesError}</p>
   410→          </div>
   411→        )}
   412→
   413→        {!courses && !coursesError && (
   414→          <div className="space-y-2">
   415→            {[0, 1, 2].map((i) => (
   416→              <div key={i} className="h-[72px] animate-pulse rounded-2xl bg-muted" />
   417→            ))}
   418→          </div>
   419→        )}
   420→
   421→        {courses && courses.length === 0 && (
   422→          <div className="rounded-2xl border bg-card p-6 text-center text-sm text-muted-foreground">
   423→            No courses yet — create one from the Home screen first.
   424→          </div>
   425→        )}
   426→
   427→        {courses && courses.length > 0 && (
   428→          <div className="space-y-3">
   429→            <div className="space-y-2" role="radiogroup" aria-label="Course">
   430→              {courses.map((c) => (
   431→                <button
   432→                  key={c.id}
   433→                  type="button"
   434→                  role="radio"
   435→                  aria-checked={selectedId === c.id}
   436→                  onClick={() => void selectCourse(c.id)}
   437→                  className={`flex w-full items-center justify-between gap-3 rounded-2xl border bg-card p-4 text-left transition-all min-h-11 ${
   438→                    selectedId === c.id
   439→                      ? 'border-primary ring-2 ring-primary/30'
   440→                      : 'hover:bg-accent/50'
   441→                  }`}
   442→                >
   443→                  <div className="min-w-0">
   444→                    <p className="font-semibold tracking-tight">{c.code}</p>
   445→                    <p className="truncate text-sm text-muted-foreground">{c.title}</p>
   446→                  </div>
   447→                  <span className="shrink-0 rounded-full bg-muted px-2.5 py-1 text-xs font-medium tabular-nums text-muted-foreground">
   448→                    {c.studentCount} students
   449→                  </span>
   450→                </button>
   451→              ))}
   452→            </div>
   453→
   454→            {selected && faceCount === 0 && (
   455→              <div className="rounded-2xl border border-amber-500/30 bg-amber-500/10 p-3 text-sm text-amber-700 dark:text-amber-400">
   456→                No students in {selected.code} have enrolled their face yet — you can
   457→                still mark students present manually during review.
   458→              </div>
   459→            )}
   460→            {selected && (faceCount ?? 0) > 0 && (
   461→              <p className="px-1 text-xs text-muted-foreground">
   462→                {faceCount} of {selected.studentCount} students face-enrolled · roster cached
   463→                for offline scanning
   464→              </p>
   465→            )}
   466→            {selected && faceCount === -1 && (
   467→              <p className="px-1 text-xs text-muted-foreground">
   468→                Roster unavailable offline — face matching may be limited.
   469→              </p>
   470→            )}
   471→
   472→            <div>
   473→              <p className="mb-1.5 px-1 text-xs font-medium uppercase tracking-wide text-muted-foreground">
   474→                Scan mode
   475→              </p>
   476→              <ToggleGroup
   477→                type="single"
   478→                value={mode}
   479→                onValueChange={(v) => v && setMode(v as SessionMode)}
   480→                className="w-full rounded-xl border bg-card p-1"
   481→              >
   482→                <ToggleGroupItem
   483→                  value="WALKTHROUGH"
   484→                  className="h-10 flex-1 gap-2 rounded-lg data-[state=on]:bg-primary data-[state=on]:text-primary-foreground"
   485→                >
   486→                  <Users className="h-4 w-4" /> Walkthrough
   487→                </ToggleGroupItem>
   488→                <ToggleGroupItem
   489→                  value="KIOSK"
   490→                  className="h-10 flex-1 gap-2 rounded-lg data-[state=on]:bg-primary data-[state=on]:text-primary-foreground"
   491→                >
   492→                  <ScanFace className="h-4 w-4" /> Kiosk
   493→                </ToggleGroupItem>
   494→              </ToggleGroup>
   495→              <p className="mt-1.5 px-1 text-xs text-muted-foreground">
   496→                {mode === 'WALKTHROUGH'
   497→                  ? 'Walk around the hall — the app recognises each student automatically.'
   498→                  : 'Students come to the camera one at a time with a head-turn check.'}
   499→              </p>
   500→            </div>
   501→          </div>
   502→        )}
   503→      </main>
   504→
   505→      <footer className="border-t bg-card px-4 pb-safe lg:px-8" style={{ paddingTop: 12, paddingBottom: 'max(env(safe-area-inset-bottom), 12px)' }}>
   506→        <Button
   507→          className="h-12 w-full text-base font-semibold"
   508→          disabled={!selectedId}
   509→          onClick={() => void startScan()}
   510→        >
   511→          Start scanning
   512→        </Button>
   513→      </footer>
   514→    </div>
   515→  )
   516→}

   598→    mirrorRef.current = facing === 'user'

   607→      })

   610→      alive = false
   611→    }
   612→  }, [])
   613→
   614→  // ---------- camera ----------
   615→  useEffect(() => {

   647→  // ---------- check-in + sync ----------

   661→            studentId: r.studentId,
   662→            name: r.name,
   663→            status: r.status,
   664→            confidence: r.confidence ?? null,
   665→            markedAt: r.markedAt,

   667→        },

   737→        ].slice(0, RECENT_MAX)

   779→        setKioskBoth({ phase: 'idle' })

   782→    [setKioskBoth]

   784→

   786→  useEffect(() => {

   794→      const canvas = canvasRef.current

   805→

   852→

   854→        if (!matchInfo) return
   855→        const entry = rosterByIdRef.current.get(matchInfo.id)
   856→        if (!entry) return

   858→          glow(entry.id)
   859→          setKioskBoth({
   860→            phase: 'info',
   861→            name: nameOf(entry),
   862→            studentId: entry.studentId,

   869→          const target: 'left' | 'right' = Math.random() < 0.5 ? 'left' : 'right'

   871→            entryId: entry.id,
   872→            target,
   873→            deadline: Date.now() + CHALLENGE_MS,

   876→            phase: 'challenge',
   877→            target,
   878→            name: entry.firstName,
   879→            nonce: Date.now(),
   880→          })
   881→        } else {
   882→          kioskCelebrate(entry, matchInfo.distance)
   883→        }
   884→        return
   885→      }

   887→      if (ui.phase === 'challenge') {
   888→        const ch = challengeRef.current
   889→        if (!ch) {
   890→          setKioskBoth({ phase: 'idle' })
   891→          return
   892→        }
   893→        if (Date.now() > ch.deadline) {
   894→          kioskIdle("Time's up — look at the camera and try again")

   898→          // a different student stepped up — restart the challenge
   899→          const entry = rosterByIdRef.current.get(matchInfo.id)
   900→          if (entry) {
   901→            const target: 'left' | 'right' = Math.random() < 0.5 ? 'left' : 'right'
   902→            challengeRef.current = {
   903→              entryId: entry.id,
   904→              target,
   905→              deadline: Date.now() + CHALLENGE_MS,
   906→            }
   907→            setKioskBoth({
   908→              phase: 'challenge',
   909→              target,
   910→              name: entry.firstName,
   911→              nonce: Date.now(),
   912→            })
   913→          }
   914→          return
   915→        }
   916→        const off = noseOffset(face)
   917→        if (
   918→          (ch.target === 'left' && off > TURN_OFFSET) ||
   919→          (ch.target === 'right' && off < -TURN_OFFSET)

   921→          const entry = rosterByIdRef.current.get(ch.entryId)
   922→          if (entry) kioskCelebrate(entry, matchInfo?.distance)

   925→      // 'success' | 'info' — just keep boxes drawn
   926→    }
   927→
   928→    const tick = (t: number) => {
   929→      raf = requestAnimationFrame(tick)

   933→      const video = videoRef.current
   934→      if (!video || video.readyState < 2) return
   935→      busyRef.current = true
   936→      const step = session.mode === 'KIOSK' ? kioskStep() : walkthroughStep()
   937→      void step
   938→        .catch(() => {})
   939→        .finally(() => {
   940→          busyRef.current = false
   941→        })
   942→    }
   943→    raf = requestAnimationFrame(tick)
   944→
   945→    return () => {
   946→      aliveRef.current = false
   947→      cancelAnimationFrame(raf)
   948→    }
   949→  }, [
   950→    cameraState,
   951→    engine,
   952→    session.mode,
   953→    liveness,
   954→    kioskIdle,
   955→    setKioskBoth,
   956→    glow,
   957→    kioskCelebrate,
   958→  ])
   959→
   960→  // ---------- lifecycle cleanup ----------
   961→  useEffect(() => {
   962→    return () => {
   963→      aliveRef.current = false
   964→      stopCamera(streamRef.current, videoRef.current)
   965→      streamRef.current = null
   966→      if (syncTimerRef.current) clearTimeout(syncTimerRef.current)
   967→      if (glowTimerRef.current) clearTimeout(glowTimerRef.current)
   968→      if (kioskTimerRef.current) clearTimeout(kioskTimerRef.current)
   969→      if (msgTimerRef.current) clearTimeout(msgTimerRef.current)
   970→      void flushSyncRef.current()
   971→    }
   972→  }, [])
   973→
   974→  // ---------- end / discard ----------
   975→  const endAndReview = useCallback(() => {
   976→    setExitOpen(false)
   977→    void flushSyncRef.current()
   978→    navigate('review', { sessionId: session.id })
   979→  }, [navigate, session.id])
   980→
   981→  const discardSession = useCallback(async () => {
   982→    setExitOpen(false)
   983→    setDiscarding(true)
   984→    try {
   985→      await api<{ ok: boolean }>(`/api/sessions/${session.id}`, { method: 'DELETE' })
   986→      for (const rec of pendingForSession(session.id)) {
   987→        removeQueued(session.id, rec.studentId)
   988→      }
   989→      useAppStore.getState().setPendingSync(pendingCount())
   990→      useAppStore.getState().setOpenSession(null)
   991→      replace('home')
   992→    } catch (e) {
   993→      toast.error(getErrorMessage(e))
   994→    } finally {
   995→      setDiscarding(false)
   996→    }
   997→  }, [replace, session.id])
   998→
   999→  const cameraUnavailable = cameraState === 'error' || cameraState === 'off'
  1000→
  1001→  // ---------- render ----------
  1002→  return (
  1003→    <div className="fixed inset-0 flex h-dvh flex-col bg-black text-white">
  1004→      {/* top bar */}
  1005→      <header className="flex items-center gap-2 bg-black/80 px-3 pt-safe" style={{ paddingTop: 'max(env(safe-area-inset-top), 10px)', paddingBottom: 10 }}>
  1006→        <Button
  1007→          variant="ghost"
  1008→          size="icon"
  1009→          className="size-10 shrink-0 text-white hover:bg-white/15"
  1010→          onClick={() => setExitOpen(true)}
  1011→          aria-label="End scanning"
  1012→        >
  1013→          <X className="h-5 w-5" />
  1014→        </Button>
  1015→        <div className="flex min-w-0 flex-1 items-center gap-2">
  1016→          <span className="rounded-full bg-white/10 px-2.5 py-1 text-xs font-semibold tracking-wide">
  1017→            {session.courseCode}
  1018→          </span>
  1019→          <span
  1020→            className={`inline-flex items-center gap-1 rounded-full px-2.5 py-1 text-xs font-bold tabular-nums ${
  1021→              checkedIds.size > 0 ? 'bg-emerald-500 text-emerald-950' : 'bg-white/10 text-white/80'
  1022→            }`}
  1023→            aria-label={`${checkedIds.size} of ${roster.length} checked in`}
  1024→          >
  1025→            <Check className="h-3.5 w-3.5" strokeWidth={3} />
  1026→            {checkedIds.size} / {roster.length}
  1027→          </span>
  1028→          {!online && (
  1029→            <span className="inline-flex items-center gap-1 rounded-full bg-amber-500/20 px-2 py-1 text-[10px] font-semibold text-amber-300">
  1030→              <WifiOff className="h-3 w-3" /> Offline — saving on device
  1031→            </span>
  1032→          )}
  1033→        </div>
  1034→        <Button
  1035→          variant="ghost"
  1036→          size="icon"
  1037→          className="size-10 shrink-0 text-white hover:bg-white/15"
  1038→          onClick={() => setFacing((f) => (f === 'user' ? 'environment' : 'user'))}
  1039→          aria-label="Flip camera"
  1040→          disabled={cameraState === 'off'}
  1041→        >
  1042→          <SwitchCamera className="h-5 w-5" />
  1043→        </Button>

  1045→
  1046→      {/* video area */}

  1049→          ref={videoRef}
  1050→          className="absolute inset-0 h-full w-full object-cover"

  1052→          playsInline
  1053→          muted
  1054→          autoPlay
  1055→        />
  1056→        <canvas ref={canvasRef} className="pointer-events-none absolute inset-0 h-full w-full" />
  1057→
  1058→        {cameraState === 'starting' && (
  1059→          <div className="absolute inset-0 flex flex-col items-center justify-center gap-3 bg-black/60">
  1060→            <span className="h-8 w-8 animate-spin rounded-full border-2 border-white/30 border-t-white" />

  1110→          <div className="no-scrollbar mb-2.5 flex gap-2 overflow-x-auto px-4" aria-live="polite">

  1115→                  initial={{ opacity: 0 }}
  1116→                  animate={{ opacity: 1 }}
