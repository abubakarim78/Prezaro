   120→  const navigate = useAppStore((s) => s.navigate)

   123→  const [phase, setPhase] = useState<'select' | 'loading' | 'live'>('select')
   124→  const [courses, setCourses] = useState<Course[] | null>(null)
   125→  const [coursesError, setCoursesError] = useState<string | null>(null)
   126→  const [settings, setSettings] = useState<AppSettings | null>(null)
   127→  const [mode, setMode] = useState<SessionMode>('WALKTHROUGH')
   128→  const [selectedId, setSelectedId] = useState<string | null>(null)
   129→  const [faceCount, setFaceCount] = useState<number | null>(null)
   130→  const [loadStage, setLoadStage] = useState('Preparing…')
   131→  const [bootError, setBootError] = useState<{ title: string; message: string } | null>(null)
   132→  const [live, setLive] = useState<{
   133→    session: Pick<SessionSummary, 'id' | 'courseId' | 'courseCode'> & { mode: SessionMode }
   134→    roster: RosterEntry[]
   135→    checkedEntryIds: string[]
   136→    recent: RecentChip[]
   137→  } | null>(null)
   138→
   139→  const rosterToken = useRef(0)
   140→  const bootToken = useRef(0)
   141→  const engineRef = useRef<FaceApi | null>(null)

   143→  // ---- courses + settings ----
   144→  useEffect(() => {
   145→    let alive = true
   146→    api<{ courses: Course[] }>('/api/courses')
   147→      .then((d) => {
   148→        if (alive) setCourses(d.courses)
   149→      })
   150→      .catch((e) => {
   151→        if (alive) setCoursesError(getErrorMessage(e))
   152→      })
   153→    api<{ settings: AppSettings }>('/api/settings')
   154→      .then((d) => {
   155→        if (alive) {
   156→          setSettings(d.settings)
   157→          setMode(d.settings.defaultMode)
   158→        }
   159→      })
   160→      .catch(() => {})
   161→    return () => {
   162→      alive = false
   163→    }
   164→  }, [])
   165→
   166→  // ---- roster count on course select ----
   167→  const selectCourse = useCallback(
   168→    async (courseId: string) => {
   169→      setSelectedId(courseId)
   170→      setFaceCount(null)
   171→      const course = courses?.find((c) => c.id === courseId)
   172→      const token = ++rosterToken.current
   173→      try {
   174→        const d = await api<{ roster: RosterEntry[] }>(`/api/courses/${courseId}/roster`)
   175→        if (rosterToken.current !== token) return
   176→        setFaceCount(d.roster.filter((r) => r.descriptors.length > 0).length)
   177→        cacheRoster({
   178→          savedAt: new Date().toISOString(),
   179→          course: {
   180→            id: courseId,
   181→            code: course?.code ?? '',
   182→            title: course?.title ?? '',
   183→          },
   184→          roster: d.roster,
   185→        })
   186→      } catch (e) {
   187→      if (rosterToken.current !== token) return
   188→        const cached = getCachedRoster(courseId)
   189→        if (cached) setFaceCount(cached.roster.filter((r) => r.descriptors.length > 0).length)
   190→        else if (e instanceof OfflineError) setFaceCount(-1)
   191→      }
   192→    },
   193→    [courses]
   194→  )
   195→
   196→  // ---- session ensure (GET open → POST) ----
   197→  const ensureSession = useCallback(
   198→    async (course: Course, sessionMode: SessionMode): Promise<SessionDetail> => {
   199→      const store = useAppStore.getState()
   200→      if (!store.online) {
   201→        const open = store.openSession
   202→        if (open && open.courseId === course.id) {
   203→          // resume the open session offline
   204→          const pending = pendingForSession(open.id)
   205→          return {
   206→            id: open.id,
   207→            courseId: course.id,
   208→            courseCode: course.code,
   209→            courseTitle: course.title,
   210→            mode: open.mode,
   211→            status: 'OPEN',
   212→            startedAt: new Date().toISOString(),
   213→            presentCount: pending.length,
   214→            rosterSize: course.studentCount,
   215→            lecturerName: store.user?.name ?? '',
   216→            records: pending.map<AttendanceRecord>((p) => ({
   217→              studentId: p.studentId,
   218→              name: p.name,
   219→              status: p.status,
   220→              confidence: p.confidence ?? null,
   221→              markedAt: p.markedAt,
   222→            })),
   223→          }
   224→        }
   225→        throw new Error('Starting a new scan needs a connection — reconnect and try again.')
   226→      }
   227→
   228→      const { sessions } = await api<{ sessions: SessionSummary[] }>(
   229→        `/api/sessions?courseId=${encodeURIComponent(course.id)}&status=OPEN`
   230→      )
   231→      let summary = sessions[0] ?? null
   232→      if (!summary) {
   233→        const created = await api<{ session: SessionSummary }>('/api/sessions', {
   234→          method: 'POST',
   235→          body: { courseId: course.id, mode: sessionMode },
   236→        })
   237→        summary = created.session
   238→      }
   239→      // Pull detail for seeding existing check-ins (best effort)
   240→      try {
   241→        const detail = await api<{ session: SessionDetail }>(`/api/sessions/${summary.id}`)
   242→        return detail.session
   243→      } catch {
   244→        return {
   245→          ...summary,
   246→          lecturerName: store.user?.name ?? '',
   247→          records: [],
   248→        }
   249→      }
   250→    },
   251→    []
   252→  )
   253→
   254→  // ---- start scanning ----
   255→  const startScan = useCallback(async () => {
   256→    const course = courses?.find((c) => c.id === selectedId)
   257→    if (!course) return
   258→    const token = ++bootToken.current
   259→    setPhase('loading')
   260→    setBootError(null)
   261→    setLoadStage('Fetching roster…')
   262→
   263→    try {
   264→      // 1. roster
   265→      let roster: RosterEntry[] | null = null
   266→      try {
   267→        const d = await api<{ course: { id: string; code: string; title: string }; roster: RosterEntry[] }>(
   268→          `/api/courses/${course.id}/roster`
   269→        )
   270→        roster = d.roster
   271→        cacheRoster({
   272→          savedAt: new Date().toISOString(),
   273→          course: d.course,
   274→          roster: d.roster,
   275→        })
   276→      } catch (e) {
   277→        const cached = getCachedRoster(course.id)
   278→        if (cached) roster = cached.roster
   279→        else if (!(e instanceof OfflineError)) throw e
   280→      }
   281→      if (bootToken.current !== token) return
   282→      if (!roster) {
   283→        throw new Error(
   284→          'You are offline and this course has no cached roster. Reconnect and try again.'
   285→        )
   286→      }
   287→
   288→      // 2. face engine
   289→      setLoadStage('Loading face engine…')
   290→      const faceapi = await loadFaceEngine((s) => {
   291→        if (bootToken.current === token) setLoadStage(s)
   292→      })
   293→      if (bootToken.current !== token) return
   294→      engineRef.current = faceapi
   295→
   296→      // 3. session (create / resume)
   297→      setLoadStage('Preparing session…')
   298→      const session = await ensureSession(course, mode)
   299→      if (bootToken.current !== token) return
   300→
   301→      // 4. seed existing check-ins (resume case)
   302→      const byStudent = new Map(roster.map((r) => [r.studentId, r]))
   303→      const checkedEntryIds = new Set<string>()
   304→      const recent: RecentChip[] = []
   305→      const records = [...session.records].sort(
   306→        (a, b) => new Date(b.markedAt).getTime() - new Date(a.markedAt).getTime()
   307→      )
   308→      for (const rec of records) {
   309→        if (rec.status === 'ABSENT') continue
   310→        const entry = byStudent.get(rec.studentId)
   311→        if (!entry) continue
   312→        checkedEntryIds.add(entry.id)
   313→        if (recent.length < 8) {
   314→          recent.push({
   315→            key: `${entry.id}-${rec.markedAt}`,
   316→            entryId: entry.id,
   317→            name: rec.name ?? nameOf(entry),
   318→            time: format(new Date(rec.markedAt), 'h:mm a'),
   319→          })
   320→        }
   321→      }
   322→
   323→      useAppStore.getState().setOpenSession({
   324→        id: session.id,
   325→        courseId: course.id,
   326→        courseCode: course.code,
   327→        mode: mode === 'KIOSK' ? 'KIOSK' : 'WALKTHROUGH',
   328→      })
   329→
   330→      setLive({
   331→        session: {
   332→          id: session.id,
   333→          courseId: course.id,
   334→          courseCode: course.code,
   335→          mode: mode === 'KIOSK' ? 'KIOSK' : 'WALKTHROUGH',
   336→        },
   337→        roster,
   338→        checkedEntryIds: [...checkedEntryIds],
   339→        recent,
   340→      })
   341→      setLoadStage('Starting camera…')
   342→      setPhase('live')
   343→    } catch (e) {
   344→      if (bootToken.current !== token) return
   345→      setBootError({
   346→        title: 'Could not start scanning',
   347→        message: getErrorMessage(e),
   348→      })
   349→    }
   350→  }, [courses, selectedId, mode, ensureSession])
   351→
   352→  // ---- render ----
   353→  if (phase === 'live' && live) {
   354→    return (
   355→      <LiveScreen
   356→        session={live.session}
   357→        roster={live.roster}
   358→        initialChecked={live.checkedEntryIds}
   359→        initialRecent={live.recent}
   360→        liveness={settings?.liveness ?? true}
   361→        engine={engineRef.current}
   362→      />
   363→    )
   364→  }
   365→
   366→  if (phase === 'loading') {
   367→    return (
   368→      <div className="flex h-dvh flex-col items-center justify-center gap-4 bg-background px-6 text-center">
   369→        <div className="relative flex h-16 w-16 items-center justify-center">
   370→          <span className="absolute inset-0 animate-ping rounded-full bg-primary/20" />
   371→          <span className="flex h-14 w-14 items-center justify-center rounded-2xl bg-primary text-primary-foreground shadow-lg shadow-primary/25">
   372→            <ScanFace className="h-7 w-7" />
   373→          </span>
   374→        </div>
   375→        <p className="text-sm font-medium text-muted-foreground" aria-live="polite">
   376→          {loadStage}
   377→        </p>
   378→        {bootError && (
   379→          <div className="w-full max-w-sm rounded-2xl border bg-card p-4 text-left">
   380→            <p className="font-semibold">{bootError.title}</p>
   381→            <p className="mt-1 text-sm text-muted-foreground">{bootError.message}</p>
   382→            <div className="mt-4 flex gap-2">
   383→              <Button className="flex-1" onClick={startScan}>
   384→                <RefreshCw className="h-4 w-4" /> Retry
   385→              </Button>
   386→              <Button variant="outline" onClick={() => setPhase('select')}>
   387→                Back
   388→              </Button>
   389→            </div>
   390→          </div>
   391→        )}
   392→      </div>
   393→    )
   394→  }
   395→
   396→  // ---- select ----
   397→  const selected = courses?.find((c) => c.id === selectedId) ?? null
   398→  return (
   399→    <div className="flex h-dvh flex-col bg-background">
   400→      <header className="flex items-center gap-3 px-4 pt-safe lg:px-8" style={{ paddingTop: 'max(env(safe-area-inset-top), 14px)', paddingBottom: 12 }}>
   401→        <Button
   402→          variant="ghost"
   403→          size="icon"
   404→          className="size-9"
   405→          onClick={() => (useAppStore.getState().openSession ? replaceHomeSafely() : useAppStore.getState().back())}
   406→          aria-label="Close scanner"
   407→        >
   408→          <X className="h-5 w-5" />
   409→        </Button>
   410→        <div>
   411→          <h1 className="text-lg font-bold tracking-tight">Take attendance</h1>
   412→          <p className="text-xs text-muted-foreground">Choose a course and scan mode</p>
   413→        </div>
   414→      </header>
   415→
   416→      <main className="flex-1 overflow-y-auto scrollbar-thin px-4 pb-6">
   417→        {coursesError && (
   418→          <div className="rounded-2xl border bg-card p-4 text-sm">
   419→            <p className="font-semibold">Couldn&apos;t load courses</p>
   420→            <p className="mt-1 text-muted-foreground">{coursesError}</p>
   421→          </div>
   422→        )}
   423→
   424→        {!courses && !coursesError && (
   425→          <div className="space-y-2">
   426→            {[0, 1, 2].map((i) => (
   427→              <div key={i} className="h-[72px] animate-pulse rounded-2xl bg-muted" />
   428→            ))}
   429→          </div>
   430→        )}
   431→
   432→        {courses && courses.length === 0 && (
   433→          <div className="rounded-2xl border bg-card p-6 text-center text-sm text-muted-foreground">
   434→            No courses yet — create one from the Home screen first.
   435→          </div>
   436→        )}
   437→
   438→        {courses && courses.length > 0 && (
   439→          <div className="space-y-3">
   440→            <div className="space-y-2" role="radiogroup" aria-label="Course">
   441→              {courses.map((c) => (
   442→                <button
   443→                  key={c.id}
   444→                  type="button"
   445→                  role="radio"
   446→                  aria-checked={selectedId === c.id}
   447→                  onClick={() => void selectCourse(c.id)}
   448→                  className={`flex w-full items-center justify-between gap-3 rounded-2xl border bg-card p-4 text-left transition-all min-h-11 ${
   449→                    selectedId === c.id
   450→                      ? 'border-primary ring-2 ring-primary/30'
   451→                      : 'hover:bg-accent/50'
   452→                  }`}
   453→                >
   454→                  <div className="min-w-0">
   455→                    <p className="font-semibold tracking-tight">{c.code}</p>
   456→                    <p className="truncate text-sm text-muted-foreground">{c.title}</p>
   457→                  </div>
   458→                  <span className="shrink-0 rounded-full bg-muted px-2.5 py-1 text-xs font-medium tabular-nums text-muted-foreground">
   459→                    {c.studentCount} students
   460→                  </span>
   461→                </button>
   462→              ))}
   463→            </div>
   464→
   465→            {selected && faceCount === 0 && (
   466→              <div className="rounded-2xl border border-amber-500/30 bg-amber-500/10 p-3 text-sm text-amber-700 dark:text-amber-400">
   467→                No students in {selected.code} have enrolled their face yet — you can
   468→                still mark students present manually during review.
   469→              </div>
   470→            )}
   471→            {selected && (faceCount ?? 0) > 0 && (
   472→              <p className="px-1 text-xs text-muted-foreground">
   473→                {faceCount} of {selected.studentCount} students face-enrolled · roster cached
   474→                for offline scanning
   475→              </p>
   476→            )}
   477→            {selected && faceCount === -1 && (
   478→              <p className="px-1 text-xs text-muted-foreground">
   479→                Roster unavailable offline — face matching may be limited.
   480→              </p>
   481→            )}
   482→
   483→            <div>
   484→              <p className="mb-1.5 px-1 text-xs font-medium uppercase tracking-wide text-muted-foreground">
   485→                Scan mode
   486→              </p>
   487→              <ToggleGroup
   488→                type="single"
   489→                value={mode}
   490→                onValueChange={(v) => v && setMode(v as SessionMode)}
   491→                className="w-full rounded-xl border bg-card p-1"
   492→              >
   493→                <ToggleGroupItem
   494→                  value="WALKTHROUGH"
   495→                  className="h-10 flex-1 gap-2 rounded-lg data-[state=on]:bg-primary data-[state=on]:text-primary-foreground"
   496→                >
   497→                  <Users className="h-4 w-4" /> Walkthrough
   498→                </ToggleGroupItem>
   499→                <ToggleGroupItem
   500→                  value="KIOSK"
   501→                  className="h-10 flex-1 gap-2 rounded-lg data-[state=on]:bg-primary data-[state=on]:text-primary-foreground"
   502→                >
   503→                  <ScanFace className="h-4 w-4" /> Kiosk
   504→                </ToggleGroupItem>
   505→              </ToggleGroup>
   506→              <p className="mt-1.5 px-1 text-xs text-muted-foreground">
   507→                {mode === 'WALKTHROUGH'
   508→                  ? 'Walk around the hall — the app recognises each student automatically.'
   509→                  : 'Students come to the camera one at a time with a head-turn check.'}
   510→              </p>
   511→            </div>
   512→          </div>
   513→        )}
   514→      </main>
   515→
   516→      <footer className="border-t bg-card px-4 pb-safe lg:px-8" style={{ paddingTop: 12, paddingBottom: 'max(env(safe-area-inset-bottom), 12px)' }}>
   517→        <Button
   518→          className="h-12 w-full text-base font-semibold"
   519→          disabled={!selectedId}
   520→          onClick={() => void startScan()}
   521→        >
   522→          Start scanning
   523→        </Button>
   524→      </footer>
   525→    </div>
   526→  )
   527→}
