   416→interface CaptureStageProps {
   417→  student: StudentDetail
   418→  poseIdx: number
   419→  setPoseIdx: (n: number) => void
   420→  capturedRef: React.MutableRefObject<Float32Array[]>
   421→  photoRef: React.MutableRefObject<string | null>
   422→  setCaptures: React.Dispatch<React.SetStateAction<Float32Array[]>>
   423→  setThumbs: React.Dispatch<React.SetStateAction<string[]>>
   424→  onCancel: () => void
   425→  onDone: () => void
   426→}
   427→
   428→function CaptureStage({
   429→  student,
   430→  poseIdx,
   431→  setPoseIdx,
   432→  capturedRef,
   433→  photoRef,
   434→  setCaptures,
   435→  setThumbs,
   436→  onCancel,
   437→  onDone,
   438→}: CaptureStageProps) {
   439→  const [engine, setEngine] = useState<FaceApi | null>(null)
   440→  const [engineStage, setEngineStage] = useState('Loading face engine…')
   441→  const [facing, setFacing] = useState<'user' | 'environment'>('user')
   442→  const [cameraState, setCameraState] = useState<'starting' | 'ready' | 'error'>('starting')
   443→  const [cameraError, setCameraError] = useState('')
   444→  const [retryKey, setRetryKey] = useState(0)
   445→  const [poseOk, setPoseOk] = useState(false)
   446→  const [faceMissing, setFaceMissing] = useState(false)
   447→  const [confirming, setConfirming] = useState(false)
   448→  const [captures, setLocalCaptures] = useState<Float32Array[]>([])
   449→  const [thumbs, setLocalThumbs] = useState<string[]>([])
   450→
   451→  const videoRef = useRef<HTMLVideoElement | null>(null)
   452→  const canvasRef = useRef<HTMLCanvasElement | null>(null)
   453→  const streamRef = useRef<MediaStream | null>(null)
   454→  const lastFaceRef = useRef<FaceResult | null>(null)
   455→  const lastSeenAtRef = useRef(0)
   456→  const poseIdxRef = useRef(poseIdx)
   457→  const busyRef = useRef(false)
   458→  const aliveRef = useRef(true)
   459→  const engineRef = useRef<FaceApi | null>(null)
   460→
   461→  useEffect(() => {
   462→    engineRef.current = engine
   463→  }, [engine])
   464→  useEffect(() => {
   465→    poseIdxRef.current = poseIdx
   466→  }, [poseIdx])
   467→
   468→  // load engine
   469→  useEffect(() => {
   470→    let alive = true
   471→    loadFaceEngine((s) => {
   472→      if (alive) setEngineStage(s)
   473→    })
   474→      .then((f) => {
   475→        if (alive) setEngine(f)
   476→      })
   477→      .catch((e) => {
   478→        if (alive) {
   479→          setEngineStage('')
   480→          setCameraError(getErrorMessage(e))
   481→          setCameraState('error')
   482→        }
   483→      })
   484→    return () => {
   485→      alive = false
   486→    }
   487→  }, [])
   488→
   489→  // camera
   490→  useEffect(() => {
   491→    if (!engine) return
   492→    let cancelled = false
   493→    let localStream: MediaStream | null = null
   494→    const video = videoRef.current
   495→    if (!video) return
   496→
   497→    // defer status update out of the synchronous effect body
   498→    Promise.resolve().then(() => {
   499→      if (!cancelled) setCameraState('starting')
   500→    })
   501→    startCamera(video, facing)
   502→      .then((s) => {
   503→        if (cancelled) {
   504→          stopCamera(s)
   505→          return
   506→        }
   507→        localStream = s
   508→        streamRef.current = s
   509→        setCameraState('ready')
   510→      })
   511→      .catch((e) => {
   512→        if (cancelled) return
   513→        setCameraError(getErrorMessage(e))
   514→        setCameraState('error')
   515→      })
   516→
   517→    return () => {
   518→      cancelled = true
   519→      if (localStream) {
   520→        stopCamera(localStream, videoRef.current)
   521→        if (streamRef.current === localStream) streamRef.current = null
   522→      }
   523→    }
   524→  }, [engine, facing, retryKey])
   525→
   526→  const grabFaceImage = useCallback(
   527→    (video: HTMLVideoElement, face: FaceResult, size: number, quality: number): string => {
   528→      const c = document.createElement('canvas')
   529→      c.width = size
   530→      c.height = size
   531→      const ctx = c.getContext('2d')
   532→      if (!ctx) return ''
   533→      // square crop around the face box with margin, mirrored for selfie view
   534→      const margin = 0.3
   535→      const crop = Math.max(face.box.width, face.box.height) * (1 + margin * 2)
   536→      const cx = face.box.x + face.box.width / 2
   537→      const cy = face.box.y + face.box.height / 2
   538→      const sx = Math.max(0, cx - crop / 2)
   539→      const sy = Math.max(0, cy - crop / 2)
   540→      ctx.translate(size, 0)
   541→      ctx.scale(-1, 1)
   542→      ctx.drawImage(video, sx, sy, crop, crop, 0, 0, size, size)
   543→      try {
   544→        return c.toDataURL('image/jpeg', quality)
   545→      } catch {
   546→        return ''
   547→      }
   548→    },
   549→    []
   550→  )
   551→
   552→  const captureCurrent = () => {
   553→    const video = videoRef.current
   554→    const face = lastFaceRef.current
   555→    if (!video || !face || busyRef.current) return
   556→    const currentPose = poseIdxRef.current
   557→    busyRef.current = true
   558→    capturedRef.current = [...capturedRef.current, face.descriptor]
   559→    setCaptures([...capturedRef.current])
   560→    setLocalCaptures([...capturedRef.current])
   561→    // straight pose → keep a larger reference photo for the student profile
   562→    if (currentPose === 0 && !photoRef.current) {
   563→      const photo = grabFaceImage(video, face, PHOTO_SIZE, 0.82)
   564→      if (photo) photoRef.current = photo
   565→    }
   566→    const thumb = grabFaceImage(video, face, THUMB_SIZE, 0.72)
   567→    if (thumb) {
   568→      setThumbs((prev) => [...prev, thumb])
   569→      setLocalThumbs((prev) => [...prev, thumb])
   570→    }
   571→    navigator.vibrate?.(40)
   572→
   573→    const advance = () => {
   574→      busyRef.current = false
   575→      setConfirming(false)
   576→      setPoseOk(false)
   577→      if (currentPose + 1 >= POSES.length) onDone()
   578→      else setPoseIdx(currentPose + 1)
   579→    }
   580→
   581→    if (currentPose === 0) {
   582→      // straight pose → grab one extra confirm descriptor 300 ms later (4 total)
   583→      setConfirming(true)
   584→      setTimeout(async () => {
   585→        try {
   586→          const v = videoRef.current
   587→          if (v && aliveRef.current && engineRef.current) {
   588→            const again = await detectSingle(engineRef.current, v)
   589→            if (again && aliveRef.current) {
   590→              capturedRef.current = [...capturedRef.current, again.descriptor]
   591→              setCaptures([...capturedRef.current])
   592→              setLocalCaptures([...capturedRef.current])
   593→            }
   594→          }
   595→        } catch {
   596→          /* best-effort confirm */
   597→        }
   598→        advance()
   599→      }, CONFIRM_DELAY_MS)
   600→    } else {
   601→      advance()
   602→    }
   603→  }
   604→
   605→  // detection loop
   606→  useEffect(() => {
   607→    aliveRef.current = true
   608→    if (!engine || cameraState !== 'ready') return
   609→    let raf = 0
   610→    let last = 0
   611→
   612→    const tick = (t: number) => {
   613→      raf = requestAnimationFrame(tick)
   614→      if (t - last < DETECT_INTERVAL_MS) return
   615→      last = t
   616→      if (busyRef.current) return
   617→      const video = videoRef.current
   618→      const canvas = canvasRef.current
   619→      if (!video || !canvas || video.readyState < 2) return
   620→      busyRef.current = true
   621→      void (async () => {
   622→        // low-light lift: detect on an enhanced canvas when the room is dim
   623→        const { source } = enhanceForDetection(video)
   624→        const face = await detectSingle(engine, source, 320)
   625→        if (!aliveRef.current) return
   626→        const mirror = true // selfie preview is mirrored
   627→        if (!face) {
   628→          drawOverlay(canvas, video, [], [], { mirror })
   629→          lastFaceRef.current = null
   630→          setFaceMissing(Date.now() - lastSeenAtRef.current > NO_FACE_HINT_MS)
   631→          setPoseOk(false)
   632→          return
   633→        }
   634→        lastFaceRef.current = face
   635→        lastSeenAtRef.current = Date.now()
   636→        setFaceMissing(false)
   637→
   638→        const vw = video.videoWidth || 1
   639→        const off = noseOffset(face)
   640→        const sizeOk = face.box.width >= SIZE_MIN * vw && face.box.width <= SIZE_MAX * vw
   641→        const centerX = face.box.x + face.box.width / 2
   642→        const centered = centerX >= CENTER_BAND * vw && centerX <= (1 - CENTER_BAND) * vw
   643→        const angleOk = POSES[poseIdxRef.current].ok(off)
   644→        const ok = sizeOk && centered && angleOk
   645→        setPoseOk(ok)
   646→        drawOverlay(
   647→          canvas,
   648→          video,
   649→          [face],
   650→          [ok ? { id: 'ok', name: '', distance: 0 } : null],
   651→          { mirror }
   652→        )
   653→      })()
   654→        .catch(() => {})
   655→        .finally(() => {
   656→          busyRef.current = false
   657→        })
   658→    }
   659→    raf = requestAnimationFrame(tick)
   660→    return () => {
   661→      aliveRef.current = false
   662→      cancelAnimationFrame(raf)
   663→    }
   664→  }, [engine, cameraState])
   665→
   666→  // cleanup
   667→  useEffect(() => {
   668→    return () => {
   669→      aliveRef.current = false
   670→      stopCamera(streamRef.current, videoRef.current)
   671→      streamRef.current = null
   672→    }
   673→  }, [])
   674→
   675→  const pose = POSES[poseIdx]
   676→  const mirror = facing === 'user'
   677→
   678→  return (
   679→    <div className="flex h-dvh flex-col bg-black text-white">
   680→      <header
   681→        className="flex items-center gap-3 bg-black/80 px-3"
   682→        style={{ paddingTop: 'max(env(safe-area-inset-top), 10px)', paddingBottom: 10 }}
   683→      >
   684→        <Button
   685→          variant="ghost"
   686→          size="icon"
   687→          className="size-10 shrink-0 text-white hover:bg-white/15"
   688→          onClick={onCancel}
   689→          aria-label="Cancel enrollment"
   690→        >
   691→          <X className="h-5 w-5" />
   692→        </Button>
   693→        <div className="min-w-0">
   694→          <p className="truncate font-semibold">{`Enroll ${student.firstName} ${student.lastName}`}</p>
   695→          <p className="text-xs tabular-nums text-white/60">
   696→            {student.studentId} · pose {poseIdx + 1} of {POSES.length}
   697→          </p>
   698→        </div>
   699→      </header>
   700→
   701→      {/* video area */}
   702→      <div className="relative min-h-0 flex-1 overflow-hidden bg-black">
   703→        <video
   704→          ref={videoRef}
   705→          className="absolute inset-0 h-full w-full object-cover"
   706→          style={{ transform: mirror ? 'scaleX(-1)' : undefined }}
   707→          playsInline
   708→          muted
   709→          autoPlay
   710→        />
   711→        <canvas ref={canvasRef} className="pointer-events-none absolute inset-0 h-full w-full" />
   712→
   713→        {cameraState === 'starting' && (
   714→          <div className="absolute inset-0 flex flex-col items-center justify-center gap-3 bg-black/70">
   715→            <span className="h-8 w-8 animate-spin rounded-full border-2 border-white/30 border-t-white" />
   716→            <p className="text-sm text-white/80">{engineStage || 'Starting camera…'}</p>
   717→          </div>
   718→        )}
   719→
   720→        {cameraState === 'error' && (
   721→          <div className="absolute inset-0 flex items-center justify-center bg-neutral-950 p-6">
   722→            <div className="w-full max-w-sm rounded-2xl border border-white/10 bg-white/5 p-5 text-center">
   723→              <div className="mx-auto flex h-12 w-12 items-center justify-center rounded-2xl bg-amber-500/15 text-amber-400">
   724→                <CameraOff className="h-6 w-6" />
   725→              </div>
   726→              <p className="mt-3 font-semibold">Camera unavailable</p>
   727→              <p className="mt-1 text-sm text-white/70">
   728→                {cameraError || 'Check the browser permission and try again.'}
   729→              </p>
   730→              <div className="mt-4 flex flex-col gap-2">
   731→                <Button
   732→                  className="h-11"
   733→                  onClick={() => {
   734→                    setCameraError('')
   735→                    setCameraState('starting')
   736→                    setRetryKey((k) => k + 1)
   737→                  }}
   738→                >
   739→                  <RefreshCw className="h-4 w-4" /> Retry
   740→                </Button>
   741→                <Button
   742→                  variant="outline"
   743→                  className="h-11 border-white/20 bg-transparent text-white hover:bg-white/10"
   744→                  onClick={onCancel}
   745→                >
   746→                  Cancel
   747→                </Button>
   748→              </div>
   749→            </div>
   750→          </div>
   751→        )}
   752→
   753→        {cameraState === 'ready' && (
   754→          <div className="absolute inset-x-0 top-0 flex flex-col items-center gap-1.5 bg-gradient-to-b from-black/70 to-transparent px-6 pb-10 pt-4 text-center">
   755→            <p className="text-xl font-bold tracking-tight">{pose.label}</p>
   756→            <p className="text-sm text-white/75">{pose.hint}</p>
   757→            {confirming && (
   758→              <motion.p
   759→                initial={{ opacity: 0 }}
   760→                animate={{ opacity: 1 }}
   761→                className="mt-1 rounded-full bg-emerald-500/20 px-3 py-1 text-xs font-semibold text-emerald-300"
   762→              >
   763→                Hold still — confirming…
   764→              </motion.p>
   765→            )}
   766→          </div>
   767→        )}
   768→      </div>
   769→
   770→      {/* bottom controls */}
   771→      <footer
   772→        className="bg-black/90"
   773→        style={{ paddingTop: 14, paddingBottom: 'max(env(safe-area-inset-bottom), 14px)' }}
   774→      >
   775→        <div className="mx-auto flex max-w-md items-center justify-between gap-4 px-6">
   776→          <div className="flex w-16 flex-col items-center gap-1">
   777→            {thumbs.length > 0 ? (
   778→               
   779→              <img
   780→                src={thumbs[thumbs.length - 1]}
   781→                alt="Last captured pose"
   782→                className="h-11 w-11 rounded-lg border border-white/20 object-cover"
   783→              />
   784→            ) : (
   785→              <span className="h-11 w-11 rounded-lg border border-dashed border-white/20" />
   786→            )}
   787→            <span className="text-[10px] tabular-nums text-white/60">
   788→              {Math.min(poseIdx + 1, POSES.length)}/{POSES.length} poses
   789→            </span>
   790→          </div>
   791→
   792→          <div className="flex flex-col items-center gap-2">
   793→            <button
   794→              type="button"
   795→              onClick={captureCurrent}
   796→              disabled={!poseOk || confirming || cameraState !== 'ready'}
   797→              aria-label="Capture pose"
   798→              className={`flex h-[72px] w-[72px] items-center justify-center rounded-full border-4 transition-all disabled:opacity-40 ${
   799→                poseOk
   800→                  ? 'border-emerald-400 bg-emerald-500 shadow-lg shadow-emerald-500/40'
   801→                  : 'border-white/40 bg-white/10'
   802→              }`}
   803→            >
   804→              <ScanFace className={`h-8 w-8 ${poseOk ? 'text-emerald-950' : 'text-white/70'}`} />
   805→            </button>
   806→            <p aria-live="polite" className="h-4 text-xs text-white/70">
   807→              {faceMissing && cameraState === 'ready'
   808→                ? 'No face — move closer / better lighting'
   809→                : poseOk
   810→                  ? 'Perfect — tap capture'
   811→                  : ''}
   812→            </p>
   813→          </div>
   814→
   815→          <div className="flex w-16 justify-end">
   816→            <Button
   817→              variant="ghost"
   818→              size="icon"
   819→              className="size-11 text-white hover:bg-white/15"
   820→              onClick={() => setFacing((f) => (f === 'user' ? 'environment' : 'user'))}
   821→              aria-label="Flip camera"
   822→            >
   823→              <SwitchCamera className="h-5 w-5" />
   824→            </Button>
   825→          </div>
   826→        </div>
   827→      </footer>
   828→    </div>
   829→  )
   830→}
