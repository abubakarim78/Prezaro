   207→            courseCode: course.code,
   208→            courseTitle: course.title,
   209→            mode: open.mode,
   210→            status: 'OPEN',

   245→          lecturerName: store.user?.name ?? '',

   276→        const cached = getCachedRoster(course.id)
   277→        if (cached) roster = cached.roster
   278→        else if (!(e instanceof OfflineError)) throw e
   279→      }
   280→      if (bootToken.current !== token) return
   281→      if (!roster) {
   282→        throw new Error(
   283→          'You are offline and this course has no cached roster. Reconnect and try again.'
   284→        )
   285→      }
   286→
   287→      // 2. face engine
   288→      setLoadStage('Loading face engine…')
   289→      const faceapi = await loadFaceEngine((s) => {

   291→      })

   293→      engineRef.current = faceapi
   294→
   295→      // 3. session (create / resume)
   296→      setLoadStage('Preparing session…')
   297→      const session = await ensureSession(course, mode)
   298→      if (bootToken.current !== token) return
   299→
   300→      // 4. seed existing check-ins (resume case)
   301→      const byStudent = new Map(roster.map((r) => [r.studentId, r]))
   302→      const checkedEntryIds = new Set<string>()
   303→      const recent: RecentChip[] = []
   304→      const records = [...session.records].sort(
   305→        (a, b) => new Date(b.markedAt).getTime() - new Date(a.markedAt).getTime()

   309→        const entry = byStudent.get(rec.studentId)
   310→        if (!entry) continue
   311→        checkedEntryIds.add(entry.id)
   312→        if (recent.length < 8) {
   313→          recent.push({
   314→            key: `${entry.id}-${rec.markedAt}`,
   315→            entryId: entry.id,
   316→            name: rec.name ?? nameOf(entry),
   317→            time: format(new Date(rec.markedAt), 'h:mm a'),
   318→          })
   319→        }
   320→      }
   321→
   322→      useAppStore.getState().setOpenSession({
   323→        id: session.id,
   324→        courseId: course.id,
   325→        courseCode: course.code,
   326→        mode: mode === 'KIOSK' ? 'KIOSK' : 'WALKTHROUGH',
   327→      })
   328→
   329→      setLive({
   330→        session: {
   331→          id: session.id,
   332→          courseId: course.id,
   333→          courseCode: course.code,
   334→          mode: mode === 'KIOSK' ? 'KIOSK' : 'WALKTHROUGH',
   335→        },
   336→        roster,
   337→        checkedEntryIds: [...checkedEntryIds],
   338→        recent,
   339→      })
   340→      setLoadStage('Starting camera…')
   341→      setPhase('live')
   342→    } catch (e) {
   343→      if (bootToken.current !== token) return
   344→      setBootError({
   345→        title: 'Could not start scanning',
   346→        message: getErrorMessage(e),
   347→      })
   348→    }
   349→  }, [courses, selectedId, mode, ensureSession])
   350→
   351→  // ---- render ----
   352→  if (phase === 'live' && live) {
   353→    return (
   354→      <LiveScreen
   355→        session={live.session}
   356→        roster={live.roster}
   357→        initialChecked={live.checkedEntryIds}
   358→        initialRecent={live.recent}
   359→        engine={engineRef.current}
   360→      />
   361→    )
   362→  }
   363→
   364→  if (phase === 'loading') {

   366→      <div className="flex h-dvh flex-col items-center justify-center gap-4 bg-background px-6 text-center">
   367→        <div className="relative flex h-16 w-16 items-center justify-center">
   368→          <span className="absolute inset-0 animate-ping rounded-full bg-primary/20" />

   370→            <ScanFace className="h-7 w-7" />
   371→          </span>
   372→        </div>
   373→        <p className="text-sm font-medium text-muted-foreground" aria-live="polite">
   374→          {loadStage}
   375→        </p>
   376→        {bootError && (
   377→          <div className="w-full max-w-sm rounded-2xl border bg-card p-4 text-left">
   378→            <p className="font-semibold">{bootError.title}</p>
   379→            <p className="mt-1 text-sm text-muted-foreground">{bootError.message}</p>
   380→            <div className="mt-4 flex gap-2">
   381→              <Button className="flex-1" onClick={startScan}>
   382→                <RefreshCw className="h-4 w-4" /> Retry
   383→              </Button>
   384→              <Button variant="outline" onClick={() => setPhase('select')}>
   385→                Back
   386→              </Button>
   387→            </div>
   388→          </div>
   389→        )}
   390→      </div>
   391→    )
   392→  }
   393→
   394→  // ---- select ----
   395→  const selected = courses?.find((c) => c.id === selectedId) ?? null
   396→  return (
   397→    <div className="flex h-dvh flex-col bg-background">
   398→      <header className="flex items-center gap-3 px-4 pt-safe lg:px-8" style={{ paddingTop: 'max(env(safe-area-inset-top), 14px)', paddingBottom: 12 }}>
   399→        <Button
   400→          variant="ghost"
   401→          size="icon"
   402→          className="size-9"
   403→          onClick={() => (useAppStore.getState().openSession ? replaceHomeSafely() : useAppStore.getState().back())}
   404→          aria-label="Close scanner"
   405→        >
   406→          <X className="h-5 w-5" />
   407→        </Button>
   408→        <div>
   409→          <h1 className="text-lg font-bold tracking-tight">Take attendance</h1>
   410→          <p className="text-xs text-muted-foreground">Choose a course and scan mode</p>
   411→        </div>
   412→      </header>
   413→
   414→      <main className="flex-1 overflow-y-auto scrollbar-thin px-4 pb-6">
   415→        {coursesError && (
   416→          <div className="rounded-2xl border bg-card p-4 text-sm">
   417→            <p className="font-semibold">Couldn&apos;t load courses</p>

   419→          </div>
   420→        )}
   421→
   422→        {!courses && !coursesError && (
   423→          <div className="space-y-2">
   424→            {[0, 1, 2].map((i) => (
   425→              <div key={i} className="h-[72px] animate-pulse rounded-2xl bg-muted" />

   427→          </div>

   429→
   430→        {courses && courses.length === 0 && (

   478→              </p>
   479→            )}
   480→
   481→            <div>
   482→              <p className="mb-1.5 px-1 text-xs font-medium uppercase tracking-wide text-muted-foreground">
   483→                Scan mode
   484→              </p>
   485→              <ToggleGroup
   486→                type="single"
   487→                value={mode}
   488→                onValueChange={(v) => v && setMode(v as SessionMode)}
   489→                className="w-full rounded-xl border bg-card p-1"
   490→              >
   491→                <ToggleGroupItem
   492→                  value="WALKTHROUGH"
   493→                  className="h-10 flex-1 gap-2 rounded-lg data-[state=on]:bg-primary data-[state=on]:text-primary-foreground"
   494→                >
   495→                  <Users className="h-4 w-4" /> Walkthrough
   496→                </ToggleGroupItem>
   497→                <ToggleGroupItem
   498→                  value="KIOSK"
   499→                  className="h-10 flex-1 gap-2 rounded-lg data-[state=on]:bg-primary data-[state=on]:text-primary-foreground"
   500→                >
   501→                  <ScanFace className="h-4 w-4" /> Kiosk
   502→                </ToggleGroupItem>
   503→              </ToggleGroup>
   504→              <p className="mt-1.5 px-1 text-xs text-muted-foreground">
   505→                {mode === 'WALKTHROUGH'
   506→                  ? 'Walk around the hall — the app recognises each student automatically.'
   507→                  : 'Students come to the camera one at a time — a single glance checks them in.'}
   508→              </p>
   509→            </div>
   510→          </div>
   511→        )}
   512→      </main>
   513→
   514→      <footer className="border-t bg-card px-4 pb-safe lg:px-8" style={{ paddingTop: 12, paddingBottom: 'max(env(safe-area-inset-bottom), 12px)' }}>
   515→        <Button
   516→          className="h-12 w-full text-base font-semibold"
   517→          disabled={!selectedId}
   518→          onClick={() => void startScan()}
   519→        >
   520→          Start scanning
   521→        </Button>
   522→      </footer>
   523→    </div>
   524→  )
   525→}
